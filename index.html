function addImageBubble(prompt) { var seed = Math.floor(Math.random() * 99999); var encoded = encodeURIComponent(prompt); var url = 'https://image.pollinations.ai/prompt/' + encoded + '?width=1024&height=1024&model=flux&nologo=true&seed=' + seed; var d = document.createElement('div'); d.className = 'msg ai'; var a = document.createElement('div'); a.className = 'av'; a.textContent = '✨'; var wrap = document.createElement('div'); wrap.className = 'imgwrap'; var loading = document.createElement('div'); loading.className = 'imgload'; loading.textContent = 'Image generate ho rahi hai... (30-60 sec)'; wrap.appendChild(loading); var attempts = 0; function tryLoad() { attempts++; var img = new Image(); img.onload = function() { loading.remove(); img.style.cursor = 'pointer'; img.onclick = function() { window.open(url, '_blank'); }; wrap.appendChild(img); chatDiv.scrollTop = chatDiv.scrollHeight; }; img.onerror = function() { if (attempts < 4) { loading.textContent = 'Retry ' + attempts + '/3... thoda wait karo'; setTimeout(tryLoad, 8000); } else { loading.innerHTML = 'Image load nahi hui. Yahan click karo'; } }; img.src = url; } setTimeout(tryLoad, 3000); d.appendChild(a); d.appendChild(wrap); chatDiv.appendChild(d); chatDiv.scrollTop = chatDiv.scrollHeight; }
